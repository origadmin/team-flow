tools.yaml 配置文件 # .team/tools.yaml
version: 1

paths:
  flow: "[$]/flow" ← flow 自身路径
  workspace: "[@]" ← workspace root
  project: "[#]" ← project root
  bin: "[$]" ← 工具二进制目录

tools:
  task:
    backend: beads
    command: "{bin}/bd"
    args_template: "{args}"
  sync:
    backend: github
    command: gh
    args_template: "issue list --repo {repo} --json"
  test:
    backend: shell
    command: go
    args_template: "test ./... -json"
  custom-report:
    backend: shell
    command: python3
    args_template: "scripts/report.py --task {task_id}"
工具发现命令 # AI 发现可用工具
flow tools list
  → task (beads) — 任务管理
  → sync (github) — 同步管理
  → test (shell) — 测试运行
  → lint (shell) — 代码检查
  → custom-report (shell) — 自定义报告

# AI 执行工具
flow tools run task list --json
  → [路由] beads
  → [替换] {bin}/bd list --json
  → [执行] 输出结果

# AI 查询工具路径
flow tools path bd
  → D:\.trae\skills\team-flow\bin\bd.exe

flow tools path flow
  → D:\.trae\skills\team-flow\bin\flow.exe
